CATEGORIES = sorted({product["category"] for product in PRODUCTS})


def get_db():
    connection = sqlite3.connect(DATABASE)
    connection.row_factory = sqlite3.Row
    return connection


def init_db():
    with get_db() as connection:
        connection.execute(
            """
            CREATE TABLE IF NOT EXISTS users (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                name TEXT NOT NULL,
                email TEXT UNIQUE NOT NULL,
                password TEXT NOT NULL,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )
            """
        )
        connection.execute(
            """
            CREATE TABLE IF NOT EXISTS orders (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                user_id INTEGER,
                order_number TEXT UNIQUE NOT NULL,
                total REAL NOT NULL,
                items TEXT NOT NULL,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                FOREIGN KEY(user_id) REFERENCES users(id)
            )
            """
        )
        connection.commit()


def money(value):
    return f"₹{float(value):,.2f}"


def get_current_user():
    user_id = session.get("user_id")
    if not user_id:
        return None

    with get_db() as connection:
        row = connection.execute("SELECT * FROM users WHERE id = ?", (user_id,)).fetchone()
        return dict(row) if row else None


def get_cart():
    cart = session.get("cart", {})
    if not isinstance(cart, dict):
        return {}
    return cart


def cart_count():
    cart = get_cart()
    return sum(int(item.get("quantity", 0)) for item in cart.values())


def get_product_by_id(product_id):
    for product in PRODUCTS:
        if product["id"] == product_id:
            return product
    return None


@app.context_processor
def inject_context():
    return {
        "current_user": get_current_user(),
        "cart_count": cart_count(),
        "money": money,
    }

@app.route("/")
@app.route("/catalog")
def catalog():
    query = (request.args.get("q") or "").strip()
    category = (request.args.get("category") or "").strip()
    product_list = PRODUCTS

    if category:
        category_key = category.lower()
        product_list = [
            product for product in product_list if product["category"].lower() == category_key
        ]

    if query:
        query_value = query.lower()
        product_list = [
            product
            for product in product_list
            if query_value in product["name"].lower()
            or query_value in product["category"].lower()
            or query_value in product["description"].lower()
        ]

    return render_template(
        "index.html",
        products=product_list,
        query=query,
        categories=CATEGORIES,
        selected_category=category,
    )


@app.route("/register", methods=["GET", "POST"])
def register():
    if request.method == "POST":
        name = (request.form.get("name") or "").strip()
        email = (request.form.get("email") or "").strip().lower()
        password = request.form.get("password") or ""
        confirm = request.form.get("confirm_password") or ""

        if not name or not email or not password:
            flash("Please complete all fields before creating an account.", "error")
            return render_template("register.html")

        if password != confirm:
            flash("Your password and confirmation do not match.", "error")
            return render_template("register.html")

        if len(password) < 6:
            flash("Password must be at least 6 characters long.", "error")
            return render_template("register.html")

        with get_db() as connection:
            existing = connection.execute(
                "SELECT id FROM users WHERE email = ?",
                (email,),
            ).fetchone()
            if existing:
                flash("An account already exists for this email address.", "error")
                return render_template("register.html")

            hashed = generate_password_hash(password)
            cursor = connection.execute(
                "INSERT INTO users (name, email, password) VALUES (?, ?, ?)",
                (name, email, hashed),
            )
            session["user_id"] = cursor.lastrowid
            session["user_name"] = name

        flash("Account created successfully. Welcome to Retail Store.", "success")
        return redirect(url_for("home"))

    return render_template("register.html")


@app.route("/login", methods=["GET", "POST"])
def login():
    if request.method == "POST":
        email = (request.form.get("email") or "").strip().lower()
        password = request.form.get("password") or ""

        if not email or not password:
            flash("Email and password are required.", "error")
            return render_template("login.html")

        with get_db() as connection:
            user = connection.execute(
                "SELECT * FROM users WHERE email = ?",
                (email,),
            ).fetchone()

        if user and check_password_hash(user["password"], password):
            session["user_id"] = user["id"]
            session["user_name"] = user["name"]
            flash("You have been logged in successfully.", "success")
            return redirect(url_for("home"))

        flash("Invalid email or password. Please try again.", "error")
        return render_template("login.html")

    return render_template("login.html")

@app.route("/add_to_cart/<int:product_id>", methods=["POST"])
def add_to_cart(product_id):
    product = get_product_by_id(product_id)
    if not product:
        flash("The selected product could not be found.", "error")
        return redirect(url_for("home"))

    cart = get_cart()
    item_id = str(product_id)
    quantity = max(1, int(request.form.get("quantity") or 1))

    existing = cart.get(item_id, {})
    existing_quantity = int(existing.get("quantity", 0))
    cart[item_id] = {
        "product_id": product["id"],
        "name": product["name"],
        "price": product["price"],
        "quantity": existing_quantity + quantity,
    }
    session["cart"] = cart

    flash(f"{product['name']} was added to your cart.", "success")
    return redirect(request.referrer or url_for("home"))


@app.route("/cart", methods=["GET", "POST"])
def cart():
    cart = get_cart()

    if request.method == "POST":
        action = request.form.get("action")
        item_id = request.form.get("item_id")

        if action == "remove" and item_id in cart:
            del cart[item_id]
            session["cart"] = cart
            flash("Item removed from your cart.", "success")
            return redirect(url_for("cart"))

        if action == "update" and item_id in cart:
            quantity = max(1, int(request.form.get("quantity") or 1))
            cart[item_id]["quantity"] = quantity
            session["cart"] = cart
            flash("Cart updated successfully.", "success")
            return redirect(url_for("cart"))

    cart_items = []
    subtotal = 0.0
    for item_id, item in cart.items():
        product = get_product_by_id(int(item_id))
        if not product:
            continue

        quantity = int(item.get("quantity", 0))
        line_total = quantity * float(item.get("price", 0))
        subtotal += line_total
        cart_items.append(
            {
                "id": item_id,
                "name": item.get("name", product["name"]),
                "price": float(item.get("price", 0)),
                "quantity": quantity,
                "line_total": line_total,
                "emoji": product.get("emoji", "🛍"),
            }
        )

    return render_template("cart.html", cart_items=cart_items, subtotal=subtotal)


@app.route("/checkout", methods=["GET", "POST"])
def checkout():
    cart = get_cart()
    if not cart:
        flash("Your cart is empty. Add a few products before checking out.", "error")
        return redirect(url_for("cart"))

    user = get_current_user()
    if not user:
        flash("Please log in to complete checkout.", "error")
        return redirect(url_for("login"))

    cart_items = []
    subtotal = 0.0
    for item_id, item in cart.items():
        product = get_product_by_id(int(item_id))
        if not product:
            continue

        quantity = int(item.get("quantity", 0))
        line_total = quantity * float(item.get("price", 0))
        subtotal += line_total
        cart_items.append(
            {
                "id": item_id,
                "name": item.get("name", product["name"]),
                "price": float(item.get("price", 0)),
                "quantity": quantity,
                "line_total": line_total,
            }
        )

    if request.method == "POST":
        order_number = f"RS-{uuid.uuid4().hex[:8].upper()}"

        with get_db() as connection:
            connection.execute(
                "INSERT INTO orders (user_id, order_number, total, items) VALUES (?, ?, ?, ?)",
                (
                    user["id"],
                    order_number,
                    round(subtotal, 2),
                    json.dumps(cart_items),
                ),
            )
            connection.commit()

        session["cart"] = {}
        flash("Demo checkout complete — no real payment was processed.", "success")
        return render_template(
            "checkout.html",
            order_number=order_number,
            cart_items=cart_items,
            subtotal=subtotal,
            success=True,
        )

    return render_template(
        "checkout.html",
        order_number=None,
        cart_items=cart_items,
        subtotal=subtotal,
        success=False,
    )
